**Step 2: 实现 `psk_for_identity/1`**

```erlang
%% @doc PSK for a CoAP-over-DTLS peer (S11-C). The PSK identity IS the
%% `users' table username, so a device presents its own name during the
%% handshake and the broker derives the key from the password it already
%% stores -- no second credential store, no certificate, nothing on the wire
%% but a derived key. Same precedent as `long_term_key/2' (TURN): the server
%% derives, the client derives, the password never crosses.
%%
%% Domain separation matters: the label plus three NUL separators make
%% `(Identity, Realm, Password)' unambiguous, and a different hash + label
%% makes a collision with TURN's MD5 derivation impossible.
%%
%% A missing user and a user without a password fail identically, so the
%% answer never reveals which names exist.
-spec psk_for_identity(binary()) -> {ok, <<_:256>>} | {error, term()}.
psk_for_identity(Identity) when is_binary(Identity) ->
    case user_record(Identity) of
        #{password := Password} when is_binary(Password) ->
            {ok, crypto:hash(sha256, [<<"unifiedmq-coaps-psk">>, 0,
                                      Identity, 0, coaps_realm(), 0, Password])};
        _ ->
            {error, bad_username_or_password}
    end;
psk_for_identity(_Other) ->
    {error, bad_username_or_password}.

coaps_realm() ->
    case application:get_env(broker_security, coaps, #{}) of
        #{realm := R} when is_binary(R), byte_size(R) > 0 -> R;
        _ -> <<"unifiedmq">>
    end.
```

加入 `-export` 列表末尾。

**Step 2: 写 `coap_dtls_acceptor.erl`**

关键形态（照 spike8 定稿，**串行 + 双路径 drain**）：

```erlang
-module(coap_dtls_acceptor).
-export([start_link/2]).

%% ListenerPid owns every established sslsocket; this process only handshakes.
%% Runs a loop, not a gen_server: no calls, no state beyond the listen socket.
start_link(ListenerPid, Opts) ->
    Pid = spawn_link(fun() -> loop(ListenerPid, Opts) end),
    {ok, Pid}.

loop(ListenerPid, #{listen := L, cfg := Cfg}) ->
    TA = maps:get(dtls_transport_accept_timeout_ms, Cfg, 1000),
    HS = maps:get(dtls_handshake_timeout_ms, Cfg, 2000),
    case catch ssl:transport_accept(L, TA) of
        {ok, S0} ->
            %% The handshake AND the identity drain happen inline, in this one
            %% process, on BOTH the success and the failure path. That is not
            %% tidiness: the lookup fun runs synchronously inside
            %% ssl:handshake and posts the identity to THIS mailbox, so a
            %% rejected identity left undrained would be picked up by the next
            %% connection (spike8's first version did exactly that).
            Ref = make_ref(),
            case catch ssl:handshake(S0, HS) of
                {ok, S} ->
                    Id = drain_identity(Ref),
                    case catch ssl:peername(S) of
                        {ok, Peer} -> admit(ListenerPid, Ref, S, Peer, Id);
                        _ -> _ = ssl:close(S)
                    end;                {error, _R} ->
                    _ = drain_identity(Ref), ok;
                {'EXIT', _R} ->
                    _ = drain_identity(Ref), ok
            end,
            loop(ListenerPid, #{listen => L, cfg => Cfg});
        {error, timeout} ->
            loop(ListenerPid, #{listen => L, cfg => Cfg});
        {error, closed} ->
            ok;
        {error, _R} ->
            loop(ListenerPid, #{listen => L, cfg => Cfg})
    end.

%% Ask the listener whether this session fits its caps, then hand the socket
%% over. `controlling_process' must be run by the current owner -- this process
%% -- and only the listener knows how many sessions exist.
%%
%% The acceptor must NOT touch the active option: ssl:setopts(S, [{active,N}])
%% from a non-owner returns `ok' and yet delivers datagrams to whoever owns the
%% socket at that moment (measured: the first N datagrams vanished with the
%% acceptor). The socket stays {active, false} until the listener arms it.
admit(ListenerPid, Ref, S, Peer, Principal) ->
    ListenerPid ! {dtls_handshake_ok, self(), Ref, S, Peer, Principal},
    receive
        {dtls_admit, Ref, ok} -> _ = ssl:controlling_process(S, ListenerPid), ok;
        {dtls_admit, Ref, refuse} -> _ = ssl:close(S), ok
    after 1000 -> _ = ssl:close(S), ok
    end.
```

**准入在 listener 单点完成**（acceptor 不能读 listener state，两份计数就是两份真相）：

```erlang
%% acceptor 侧，握手成功后：
Ref = make_ref(),
ListenerPid ! {dtls_handshake_ok, self(), Ref, S, Peer, Principal},
receive
    {dtls_admit, Ref, ok}     -> ok = ssl:controlling_process(S, ListenerPid), ok;
    {dtls_admit, Ref, refuse} -> _ = ssl:close(S), ok
after 1000 -> _ = ssl:close(S), ok      %% listener 忙/已死：按拒绝处理
end
```

listener 侧：`max_dtls_sessions`（`peers` 表大小）与 `dtls_max_sessions_per_principal`（`principal_sessions` 计数）检查通过 → 回 `{dtls_admit, Ref, ok}`、建 `peers`/`socks`/`idents` 三表并 `ssl:setopts(S, [{active, ?ACTIVE_N}])`；否则回 `refuse`。

**启动顺序（不能倒）**：acceptor 先 `spawn_link`（它 `receive {listen_socket, L}` 阻塞等），`Ref = make_ref()`；listener 用

```erlang
{user_lookup_fun, {fun(psk, Id, {Pid, Ref0}) ->
                          Pid ! {psk_identity, Ref0, Id},
                          case broker_security:psk_for_identity(Id) of
                              {ok, Psk} -> {ok, Psk};
                              {error, _} -> ?ALERT_REC(?FATAL, ?UNKNOWN_PSK_IDENTITY)
                          end
                  end, {AcceptorPid, Ref}}}
```

建 `ssl:listen`，成功后 `AcceptorPid ! {listen_socket, L}`。这与 spike8 的顺序一致，也是 `psk_identity` 能落到 acceptor 邮箱的唯一办法（lookup fun 的 State 在 `ssl:listen` 时固定，抓不到之后才启动的 pid）。

拒绝必须返回 `#alert{}`（spike10 实测：客户端 ~224 ms 得 `{tls_alert,{handshake_failure,_}}`）；返回 `{error, _}` 在 spike8 里表现为客户端**超时**（不干净）。因此 `coap_listener` 需要 `-include_lib("ssl/src/ssl_alert.hrl").`（OTP 自带头文件，不是新依赖）。

`drain_identity(Ref)` 只匹配 `{psk_identity, Ref, Id}`（`after 0 -> undefined`），因此不会误取别的连接的消息——**双路径 drain 仍是必需的**，因为被拒连接的 identity 消息同样会到达。

**Step 3: listener 的 DTLS 分支**

`#st{}` 新增字段：

```erlang
transport = udp :: udp | dtls,
peers = #{} :: map(),       %% SslSock => {IP,Port}
socks = #{} :: map(),       %% {IP,Port} => SslSock
idents = #{} :: map(),      %% {IP,Port} => #{principal := binary(), tenant := binary()}
```

`init/1`：`case maps:get(transport, Cfg, udp) of udp -> <现状>; dtls -> init_dtls(Cfg) end`。`init_dtls`：

1. `{ok,_} = application:ensure_all_started(ssl)`；
2. 空 `users` 表 → `{stop, {dtls_no_credentials}}`（设计 §4.3）；
3. `Suites = cert_free_psk_suites()`（helper，筛 `anonymous` + `psk|dhe_psk|ecdhe_psk` + AEAD），**空则** `{stop, {dtls_no_suites}}`；
4. 顺序不能倒（见上）：`spawn_link` acceptor（它阻塞等 `{listen_socket, L}`）→ 用捕获 `{AcceptorPid, Ref}` 的 `user_lookup_fun` 建 `ssl:listen` → `AcceptorPid ! {listen_socket, L}`；
5. `ActualPort` 从 `ssl:sockname(L)` 取（`port/1` 仍回 `{ok, Port}`）。

新增 `handle_info` 子句：

```erlang
handle_info({dtls_handshake_ok, From, Ref, S, Peer, Principal}, St) ->
    case admit_session(Principal, Peer, St) of
        {ok, St1} ->
            From ! {dtls_admit, Ref, ok},
            _ = ssl:setopts(S, [{active, ?ACTIVE_N}]),
            {noreply, St1};
        refuse ->
            From ! {dtls_admit, Ref, refuse},
            {noreply, St}
    end;
handle_info({ssl, S, Bin}, #st{peers = Peers} = St) ->
    case maps:find(S, Peers) of
        {ok, {IP, Port}} -> {noreply, on_packet(IP, Port, Bin, St)};
        error -> {noreply, St}     %% closed race: socket we no longer track
    end;
handle_info({ssl_closed, S}, St) -> {noreply, drop_session(S, St)};
handle_info({ssl_error, S, Reason}, St) -> logger:warning(...), {noreply, drop_session(S, St)};
handle_info({ssl_passive, S}, St) -> re-arm {active, ?ACTIVE_N} on S;
handle_info({'EXIT', Acceptor, Reason}, #st{dtls_acceptor = Acceptor} = St) ->
    {stop, {dtls_acceptor_died, Reason}, St};
```

```erlang
start(_Type, _Args) ->
    case {enabled(), guard()} of
        {false, _} -> broker_proto_coap_sup:start_link();
        {true, ok} -> broker_proto_coap_sup:start_link();
        {true, {error, Reason}} -> {error, Reason}
    end.

%% `coap' (plain UDP) can never carry a credential, so requiring auth for it is
%% refused. `coaps' (DTLS) can, but only if EVERY listener is actually DTLS --
%% a single plain listener would silently downgrade the deployment to an
%% anonymous front-end, which is exactly the failure §D5 exists to prevent.
guard() ->
    Required = require_auth_list(),
    case lists:member(<<"coap">>, Required) of
        true -> {error, {require_auth_unsupported, coap}};
        false ->
            case lists:member(<<"coaps">>, Required) of
                false -> ok;
                true ->
                    case lists:all(fun is_dtls/1, listeners()) of
                        true -> ok;
                        false -> {error, {require_auth_needs_dtls, coaps}}
                    end
            end
    end.
```
